```typescript
// health.service.ts
@Injectable()
export class HealthService {
  constructor(
    @InjectRepository(User) private readonly repo: Repository<User>,
    @InjectRepository(Session) private readonly sessionRepo: Repository<Session>,
  ) {}

  async check() {
    let dbStatus = 'ok';
    let lastMigration: string | null = null;
    try {
      await this.repo.query('SELECT 1');
      const migrations = await this.repo.query(
        `SELECT name FROM migrations ORDER BY timestamp DESC LIMIT 1`,
      );
      lastMigration = migrations[0]?.name ?? null;
    } catch {
      dbStatus = 'error';
    }

    return {
      status: dbStatus === 'ok' ? 'ok' : 'degraded',
      timestamp: new Date().toISOString(),
      uptime: process.uptime(),
      memory: {
        heapUsed: process.memoryUsage().heapUsed,
        heapTotal: process.memoryUsage().heapTotal,
        rss: process.memoryUsage().rss,
      },
      database: dbStatus,
      lastMigration,
    };
  }
}
```

### Work Item 1.4: Fix EmailModule Timing Bug (MAJ-09)

**Goal:** Fix `EmailModule.forRoot()` evaluating `process.env.NODE_ENV` at module import time instead of runtime. In test environments where `NODE_ENV` isn't set until after module imports, this causes `ResendEmailService` to be selected instead of `DevEmailService`.

**Root cause:** Dynamic modules execute at import time. When `EmailModule.forRoot()` runs during `AppModule` construction, `process.env.NODE_ENV` may not yet reflect the test environment. The class selection happens once and is frozen.

**Fix:** Replace `useClass` with `useFactory` that resolves `ConfigService` at provider initialization time. ConfigService reads from `@nestjs/config` which loads `.env` files and environment variables before providers are instantiated.

#### Files to Modify

| File | Required Changes |
|------|-----------------|
| `backend/src/email/email.module.ts` | Change `forRoot()` from `isProduction ? ResendEmailService : DevEmailService` pattern to a factory provider that injects `ConfigService` and checks `configService.get('NODE_ENV') === 'production'` |

#### Implementation

```typescript
// email.module.ts — fixed forRoot()
@Module({})
export class EmailModule {
  static forRoot(): DynamicModule {
    return {
      module: EmailModule,
      providers: [
        ResendEmailService,
        DevEmailService,
        {
          provide: 'EMAIL_SERVICE',
          useFactory: (configService: ConfigService) => {
            return configService.get('NODE_ENV') === 'production'
              ? new ResendEmailService(configService)
              : new DevEmailService();
          },
          inject: [ConfigService],
        },
      ],
      exports: ['EMAIL_SERVICE'],
    };
  }
}
```

Note: `ResendEmailService` already has `OnModuleInit` that validates the API key. The factory creates an instance directly — the lifecycle hook still fires because NestJS manages the instance registered via `ResendEmailService` provider.

| EmailModule uses resend in prod | Mock NODE_ENV=production, verify ResendEmailService is provided | `pnpm test -- email` |

### Work Item 1.5: Fix Health Module Entity Coupling (MIN-07)

**Goal:** Remove `HealthService` dependency on `User` entity. Currently injects `@InjectRepository(User)` just to run `SELECT 1` and query the `migrations` table. This creates an unnecessary coupling that breaks if the User entity changes or is removed.

**Fix:** Replace `@InjectRepository(User)` with `DataSource` injection. `DataSource` provides a raw query interface without coupling to any specific entity.

#### Files to Modify

| File | Required Changes |
|------|-----------------|
| `backend/src/health/health.service.ts` | Replace `@InjectRepository(User) private readonly repo` with `@InjectDataSource() private readonly dataSource: DataSource`. Change `this.repo.query(...)` → `this.dataSource.query(...)`. Remove `InjectRepository` and `Repository` imports. |
| `backend/src/health/health.module.ts` | Replace `TypeOrmModule.forFeature([User])` with `TypeOrmModule.forFeature([])`. Import `TypeOrmModule` from `@nestjs/typeorm`. |

#### Implementation

```typescript
// health.service.ts — fixed

import { DataSource } from 'typeorm';

@Injectable()
```

### Work Item 1.6: Fix forgotPassword HTTP Status Code (MIN-05)

**Goal:** Currently `POST /auth/forgot-password` returns `201 Created` by default (Post method). Should return `200 OK` since the operation doesn't create a new resource — it sends an email.

**Fix:** Add `@HttpCode(HttpStatus.OK)` to the `forgotPassword()` method in `AuthController`.

#### Files to Modify

| File | Required Changes |
|------|-----------------|
| `backend/src/auth/auth.controller.ts` | Add `@HttpCode(HttpStatus.OK)` decorator to `forgotPassword()` method. Import `HttpCode` from `@nestjs/common` and `HttpStatus` from `@nestjs/common`. |

#### Implementation

```typescript
@Post('forgot-password')
@HttpCode(HttpStatus.OK)
@Throttle({ default: { limit: 1, ttl: 60000 } })
async forgotPassword(@Body() dto: ForgotPasswordDto) {
  await this.authService.forgotPassword(dto.email);
}
```
